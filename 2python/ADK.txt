example-1 [create Agent with google model gemini-2.0-flash]
	create folder(name anyything)
	create sub folder this folder name should be agent name
	__init__.py
	.env file
		key-this is created in google ai studio for google model
	agent.py- this our class
		in this call import Google agent class and pass parameters like name, model, description, instructions
		google model details we can find here https://ai.google.dev/gemini-api/docs/models
	from google.adk.agents import Agent
	from google.adk.runners import Runner

	# Define the agent
	qa_agent = Agent(
		name="qa_agent",
		model="gemini-1.5-pro",
		instruction="""
		You are a helpful assistant.
		Your job is to answer user questions clearly and concisely.
		
		Guidelines:
		- If the user asks a factual question, provide a clear explanation.
		- If the question is unclear, ask for clarification.
		- Use simple language and short examples when helpful.
		"""
	)
	run command > adk web
